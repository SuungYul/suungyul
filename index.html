# 공개용 재구성: DB 연결의 반환 책임을 한곳에 둡니다.
from contextlib import contextmanager

@contextmanager
def connection_scope(acquire, release):
    connection = acquire()
    try:
        yield connection
    finally:
        release(connection)

def fetch_by_id(acquire, release, item_id):
    with connection_scope(acquire, release) as conn:
        with conn.cursor() as cursor:
            cursor.execute(
                "SELECT id FROM items WHERE id = %s",
                (item_id,),
            )
            return cursor.fetchone()

# 공개용 재구성: [시작, 끝) 범위를 월별로 나눕니다.
from datetime import date

def month_windows(start: date, stop: date):
    if start >= stop:
        raise ValueError("start must precede stop")
    cursor = start
    while cursor < stop:
        next_month = (
            date(cursor.year + 1, 1, 1)
            if cursor.month == 12
            else date(cursor.year, cursor.month + 1, 1)
        )
        end = min(next_month, stop)
        yield cursor, end
        cursor = end
